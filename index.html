Code Arduino pour connexion

Copiez ce code dans l'IDE Arduino et téléversez-le sur votre carte ESP8266/ESP32.

#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>
#include <Servo.h>

const char* ssid = "VotreSSID";
const char* password = "VotreMotDePasse";

ESP8266WebServer server(80);
Servo servo;

void setup() {
  pinMode(LED_BUILTIN, OUTPUT);
  Serial.begin(115200);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
  }
  Serial.println(WiFi.localIP());
  server.on("/led", handleLed);
  server.on("/execute", HTTP_POST, handleExecute);
  server.begin();
}

void loop() {
  server.handleClient();
}

void handleLed() {
  String status = server.arg("status");
  if (status == "ON") {
    digitalWrite(LED_BUILTIN, LOW);
  } else {
    digitalWrite(LED_BUILTIN, HIGH);
  }
  server.send(200, "text/plain", "OK");
}

void handleExecute() {
  String code = server.arg("plain");
  Serial.println("Code reçu: " + code);
  
  // Parser et exécuter les commandes ligne par ligne
  int start = 0;
  int end = code.indexOf('\n');
  while (end != -1) {
    String line = code.substring(start, end);
    executeCommand(line);
    start = end + 1;
    end = code.indexOf('\n', start);
  }
  if (start < code.length()) {
    executeCommand(code.substring(start));
  }
  
  server.send(200, "text/plain", "Code exécuté");
}

void executeCommand(String cmd) {
  cmd.trim();
  if (cmd.startsWith("led ")) {
    int space1 = cmd.indexOf(' ', 4);
    int pin = cmd.substring(4, space1).toInt();
    String state = cmd.substring(space1 + 1);
    pinMode(pin, OUTPUT);
    digitalWrite(pin, state == "on" ? LOW : HIGH);
  } else if (cmd.startsWith("servo ")) {
    int space1 = cmd.indexOf(' ', 6);
    int pin = cmd.substring(6, space1).toInt();
    int angle = cmd.substring(space1 + 1).toInt();
    servo.attach(pin);
    servo.write(angle);
  } else if (cmd.startsWith("delay ")) {
    int ms = cmd.substring(6).toInt();
    delay(ms);
  } else if (cmd.startsWith("tone ")) {
    int space1 = cmd.indexOf(' ', 5);
    int pin = cmd.substring(5, space1).toInt();
    int freq = cmd.substring(space1 + 1).toInt();
    tone(pin, freq);
  } else if (cmd.startsWith("noTone ")) {
    int pin = cmd.substring(7).toInt();
    noTone(pin);
  } else if (cmd.startsWith("analogWrite ")) {
    int space1 = cmd.indexOf(' ', 11);
    int pin = cmd.substring(11, space1).toInt();
    int val = cmd.substring(space1 + 1).toInt();
    analogWrite(pin, val);
  }
  // Ajoutez d'autres commandes ici
}

Modifiez le SSID et le mot de passe, puis téléversez. Notez l'IP dans le moniteur série.

❓ Aide

📁 Configuration du dossier projet

Pour sauvegarder vos projets :

  1. Allez dans l'onglet Projects
  2. Cliquez sur "📁 Configurer le dossier projet"
  3. Sélectionnez le dossier projet/ de votre dépôt
  4. Autorisez l'accès au dossier

💡 Important : Cette fonctionnalité fonctionne uniquement sur Chrome ou Edge.

💾 Synchronisation Git

Vos projets sont automatiquement sauvegardés dans le dossier projet/.

Pour synchroniser avec GitHub :

  1. Ouvrez un terminal dans votre dépôt
  2. Exécutez les commandes : 
    git add projet/*.json
git commit -m "Sauvegarde projets"
git push

📄 Dupliquer un projet

Pour dupliquer un projet existant :

  1. Ouvrez le projet à dupliquer
  2. Cliquez sur le bouton "📋 Dupliquer" à gauche du bouton OK
  3. Une copie sera créée avec le suffixe "(copie)"

📅 Timeline des projets

Les projets sont automatiquement triés par date de dernière modification (le plus récent en premier).

La date affichée sous chaque projet correspond à la dernière sauvegarde.

⚡ Composants

Pour ajouter un composant à un projet :

  1. Ouvrez le projet
  2. Cliquez sur "+ AJOUTER UN COMPOSANT"
  3. Sélectionnez une catégorie, puis le composant
  4. Le composant sera ajouté à votre projet avec ses informations techniques

🧮 Formules

L'onglet Formula contient des calculateurs pour :

  • Loi d'Ohm (V = R × I)
  • Puissance (électrique, mécanique)
  • Capacité et fréquence
  • Diviseur de tension
